// 1. Maak het hoofdvenster (figure)
f = figure("figure_name", "Mijn Scilab Interface", ...
           "position", [200, 200, 400, 300], ...
           "backgroundcolor", [0.95, 0.95, 0.95]);

// 2. Voeg een tekstlabel toe
lbl = uicontrol(f, "style", "text", ...
                "string", "Klaar voor berekeningen", ...
                "position", [100, 200, 200, 40], ...
                "fontsize", 14, ...
                "backgroundcolor", [0.95, 0.95, 0.95], ...
                "horizontalalignment", "center");

// 3. Voeg een interactieve knop toe
// De 'callback' bepaalt wat er gebeurt als je klikt
btn = uicontrol(f, "style", "pushbutton", ...
                "string", "Teken een Functie", ...
                "position", [125, 100, 150, 40], ...
                "fontsize", 12, ...
                "callback", "clf(f); plot2d(); disp(''Standaard plot is gegenereerd!'')");
